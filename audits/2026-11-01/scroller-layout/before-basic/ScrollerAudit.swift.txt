import AppKit
import FluentKit

@main struct ScrollerAudit {
    @MainActor static func main() throws {
        _ = NSApplication.shared; NSApp.setActivationPolicy(.accessory); NSApp.finishLaunching()
        func all(_ v: NSView) -> [NSView] { [v] + v.subviews.flatMap(all) }
        func settle() { RunLoop.main.run(until: Date(timeIntervalSinceNow: 0.12)) }
        func rect(_ r: NSRect) -> [CGFloat] { [r.minX, r.minY, r.width, r.height] }
        func rows() -> some FluentView {
            FluentVStack(spacing: 0) { FluentForEach(Array(0..<60), id: \.self) { FluentText("Row \($0)").frame(height: 32) } }
        }
        var results: [[String: Any]] = []
        for rtl in [false, true] {
            for kind in ["scroll", "list", "listview", "grid", "tree", "refresh", "refresh-horizontal", "autosuggest", "combo", "annotated"] {
                let query = FluentState(wrappedValue: "")
                let selection = FluentState<Int?>(wrappedValue: 0)
                let content: FluentAnyView
                switch kind {
                case "scroll": content = FluentAnyView(FluentScrollView { rows() })
                case "list": content = FluentAnyView(FluentList(rows: (0..<100).map { FluentText("Row \($0)") }, rowHeight: 32))
                case "listview": content = FluentAnyView(FluentListView(Array(0..<100), rowHeight: 32) { FluentText("Row \($0)") })
                case "grid": content = FluentAnyView(FluentCollection(snapshot: FluentCollectionSnapshot(section: 0, items: Array(0..<100)),
                    layout: .adaptiveGrid(minimumItemWidth: 120, itemHeight: 60, spacing: 8), content: { FluentText("Row \($0)") }))
                case "tree": content = FluentAnyView(FluentTreeView(nodes: (0..<100).map { index in FluentTreeViewNode(id: index) { FluentText("Row \(index)") } }))
                case "refresh": content = FluentAnyView(FluentRefreshContainer(onRefreshRequested: { $0.complete() }) { rows() })
                case "refresh-horizontal": content = FluentAnyView(FluentRefreshContainer(pullDirection: .leftToRight, onRefreshRequested: { $0.complete() }) {
                    FluentHStack { FluentForEach(Array(0..<20), id: \.self) { FluentText("Column \($0)").frame(width: 80) } }
                })
                case "autosuggest": content = FluentAnyView(FluentAutoSuggestBox(query.projectedValue,
                    suggestions: { _ in (0..<30).map { .init(id: $0, title: "Result \($0)") } }, maxSuggestionListHeight: 180))
                case "combo": content = FluentAnyView(FluentComboBox(options: Array(0..<30), selection: selection.projectedValue))
                default: content = FluentAnyView(FluentAnnotatedScrollView(labels: [.init("Start", scrollOffset: 0)]) { rows() })
                }
                let root = FluentViewHost(content.frame(width: 420, height: 300), context: .init(reduceMotion: true, layoutDirection: rtl ? .rightToLeft : .leftToRight))
                let window = NSWindow(contentRect: NSRect(x: 100, y: 360, width: 420, height: 300), styleMask: [.titled, .resizable], backing: .buffered, defer: false)
                window.isReleasedWhenClosed = false
                let stage = NSView(frame: NSRect(x: 0, y: 0, width: 420, height: 300)); window.contentView = stage
                root.frame = stage.bounds; root.autoresizingMask = [.width, .height]; stage.addSubview(root)
                window.makeKeyAndOrderFront(nil); root.layoutSubtreeIfNeeded(); settle()
                if kind == "autosuggest" {
                    let field = all(root).first { $0.identifier?.rawValue == "FluentKit.AutoSuggestBox.Field" } as! NSTextField
                    field.selectText(nil); (field.currentEditor() as! NSTextView).insertText("Ap", replacementRange: NSRange(location: 0, length: 0))
                    RunLoop.main.run(until: Date(timeIntervalSinceNow: 0.25))
                } else if kind == "combo" {
                    all(root).compactMap { $0 as? NSComboBox }.first!.performClick(nil); settle()
                }
                let surfaces = [root as NSView] + (window.childWindows ?? []).compactMap(\.contentView)
                let scroll = surfaces.flatMap(all).compactMap { $0 as? NSScrollView }.first!
                precondition(scroll.frame.width > 0 && scroll.frame.height > 0, "Probe must have a visible \(kind) viewport")
                scroll.scrollerStyle = .overlay
                scroll.superview?.needsLayout = true; surfaces.forEach { $0.layoutSubtreeIfNeeded() }; settle()
                scroll.contentView.scroll(to: kind == "refresh-horizontal" ? NSPoint(x: 100, y: 0) : NSPoint(x: 0, y: 100))
                scroll.reflectScrolledClipView(scroll.contentView); settle()
                for (step, style) in [NSScroller.Style.overlay, .legacy, .overlay, .legacy].enumerated() {
                    scroll.scrollerStyle = style
                    // No parent layout request or window resize: AppKit itself must propagate the new viewport.
                    settle()
                    let document = scroll.documentView!
                    var row: [String: Any] = ["kind": kind, "rtl": rtl, "step": step, "style": style.rawValue,
                        "viewport": rect(scroll.contentView.bounds), "document": rect(document.frame), "scroll": rect(scroll.frame),
                        "verticalVisible": scroll.verticalScroller.map { !$0.isHidden } ?? false,
                        "horizontalVisible": scroll.horizontalScroller.map { !$0.isHidden } ?? false,
                        "childMaxX": document.subviews.map(\.frame.maxX).max() ?? 0,
                        "parentNeedsLayout": scroll.superview?.needsLayout ?? false]
                    if let collection = document as? NSCollectionView, let layout = collection.collectionViewLayout as? NSCollectionViewFlowLayout {
                        row["itemWidth"] = layout.itemSize.width
                        row["visibleItems"] = collection.visibleItems().count
                    }
                    if let table = document as? NSTableView { row["columnWidths"] = table.tableColumns.map(\.width) }
                    results.append(row)
                }
                window.close()
            }
        }
        print(String(decoding: try JSONSerialization.data(withJSONObject: results, options: [.prettyPrinted, .sortedKeys]), as: UTF8.self))
    }
}
